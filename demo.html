// 片内硬件组成

掌握AT89S52 片内资源

  • CPU：8位微处理器
  • ROM：8KB Flash 程序存储器
  • RAM：256B 数据存储器
  • I/O口：P0、P1、P2、P3 四个8位并行口
  • 定时器：T0、T1、T2 三个16位
  • 中断源：5个（INT0、T0、INT1、T1、串口）
  • 串行口：UART 全双工
出题角度：简答题"51单片机片内包含哪些功能部件？"

// 并行I/O口

掌握四个并行口对比

端口类型第二功能外扩时作用
P0开漏输出无低8位地址/数据复用
P1准双向口无—
P2准双向口无高8位地址
P3准双向口有—

掌握P3口第二功能（常考）

  • P3.0 — RXD 串行接收
  • P3.1 — TXD 串行发送
  • P3.2 — INT0 外部中断0
  • P3.3 — INT1 外部中断1
  • P3.4 — T0 定时器0输入
  • P3.5 — T1 定时器1输入
  • P3.6 — WR 外部RAM写
  • P3.7 — RD 外部RAM读

// 存储结构

理解片内RAM空间（256B）

地址区域说明
00H-1FH工作寄存器区4组 × 8个（R0-R7），RS1/RS0选择
20H-2FH位寻址区128个位地址（00H-7FH）
30H-7FH通用RAM区数据缓冲区、堆栈
80H-FFHSFR区特殊功能寄存器
出题重点：四个区域的地址范围和功能，特别是位寻址区

// 定时器初值计算

掌握方式1（16位）初值公式

TH0 = (65536 - T × fosc / 12) / 256

TL0 = (65536 - T × fosc / 12) % 256

示例：fosc=12MHz，欲定时50ms

TH0 = (65536 - 0.05 × 12×10⁶ / 12) / 256
= (65536 - 50000) / 256 = 15536 / 256 ≈ 60 → 0x3C
TL0 = 15536 % 256 = 176 → 0xB0

// 流水灯（三种方法）

📄 方法一：循环移位

#include <reg51.h>
void delay(unsigned int ms) {
    unsigned int i, j;
    for (i = ms; i > 0; i--)
        for (j = 120; j > 0; j--);
}

void main() {
    unsigned char i;
    while (1) {
        P1 = 0xFE;  // 1111 1110
        for (i = 0; i < 8; i++) {
            delay(500);
            P1 = (P1 << 1) | 0x01;
        }
    }
}

📄 方法二：数组查表

#include <reg51.h>
void delay(unsigned int ms);

unsigned char code table[] = {
    0xFE, 0xFD, 0xFB, 0xF7,
    0xEF, 0xDF, 0xBF, 0x7F
};

void main() {
    unsigned char i;
    while (1) {
        for (i = 0; i < 8; i++) {
            P1 = table[i];
            delay(500);
        }
    }
}

// 外部中断

📄 INT0 下降沿触发

#include <reg51.h>
sbit LED = P1^0;

void main() {
    EA = 1;      // 总中断允许
    EX0 = 1;     // 允许INT0中断
    IT0 = 1;     // 下降沿触发
    while (1);
}

void Int0_ISR() interrupt 0 {
    LED = !LED;  // 取反LED
}

// 数码管动态扫描

📄 共阳极数码管显示"0123"

#include <reg51.h>
// 共阳极段码表：0-9
unsigned char code seg[] = {
    0xC0, 0xF9, 0xA4, 0xB0,
    0x99, 0x92, 0x82, 0xF8,
    0x80, 0x90
};

void delay(unsigned int ms);

void main() {
    while (1) {
        P2 = 0x01;  // 位选第1位
        P0 = seg[0]; // 显示"0"
        delay(5);

        P2 = 0x02;  // 位选第2位
        P0 = seg[1]; // 显示"1"
        delay(5);

        P2 = 0x04;  // 位选第3位
        P0 = seg[2]; // 显示"2"
        delay(5);

        P2 = 0x08;  // 位选第4位
        P0 = seg[3]; // 显示"3"
        delay(5);
    }
}
